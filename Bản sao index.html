<script type="module">

import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js";

import { getDatabase, ref, set, onValue } from "https://www.gstatic.com/firebasejs/10.12.0/firebase-database.js";


/* Danh sách địa chỉ Firebase Realtime Database tự động dò tìm cho dự án theodoivanhanhxe */

const FIREBASE_URLS = [

  "https://theodoivanhanhxe-default-rtdb.asia-southeast1.firebasedatabase.app",

  "https://theodoivanhanhxe-default-rtdb.firebaseio.com"

];


const VEHICLES=[...[1,2].map(n=>({id:'RS-0'+n,type:'RS',label:'0'+n})),...Array.from({length:15},(_,i)=>({id:'FL-'+String(i+1).padStart(2,'0'),type:'FL',label:String(i+1)}))];

const full=id=>{const v=VEHICLES.find(x=>x.id===id);return (v.type==='RS'?'Cẩu gắp ':'Xe nâng ')+v.label};

const ST={running:'Đang vận hành',idle:'Chờ lệnh',maint:'Bảo dưỡng / sự cố',off:'Không hoạt động'};

const COL={running:'var(--run)',idle:'var(--idle)',maint:'var(--mt)',off:'var(--off)'};

const state={};let store=null,filter='all',cur=null;

const $=id=>document.getElementById(id);

const esc=s=>String(s??'').replace(/[&<>"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));

const get=id=>state[id]||{status:'off',operator:'',task:'',since:0,fuel:100,hours:0,log:[]};

const live=d=>(d.hours||0)+(d.status==='running'&&d.since?(Date.now()-d.since)/36e5:0);

const dur=ms=>{const m=Math.floor(ms/6e4);return m<1?'vừa xong':m<60?m+' phút':Math.floor(m/60)+'g '+(m%60)+'p'};

const init=n=>n.trim().split(/\s+/).slice(-2).map(w=>w[0]).join('').toUpperCase()||'?';

const fmt=t=>new Date(t).toLocaleString('vi-VN',{day:'2-digit',month:'2-digit',hour:'2-digit',minute:'2-digit'});


function render(){

  const cnt={running:0,idle:0,maint:0,off:0},low=[];

  VEHICLES.forEach(v=>{const d=get(v.id);cnt[d.status]++;if(d.fuel<=20&&d.status!=='off')low.push(full(v.id))});

  const N=VEHICLES.length;

  $('ops').innerHTML=[...new Set(Object.values(state).map(d=>d&&d.operator).filter(Boolean))].map(n=>`<option>${esc(n)}</option>`).join('');

  $('stats').innerHTML=[['all','Tổng số xe',N],['running','Đang vận hành',cnt.running],['idle','Chờ lệnh',cnt.idle],['maint','Bảo dưỡng / sự cố',cnt.maint],['off','Không hoạt động',cnt.off]]

    .map(([k,l,n])=>`<div class="stat ${filter===k?'on':''}" data-f="${k}"><b>${n}</b><span>${l}</span></div>`).join('');

  $('bar').innerHTML=Object.keys(ST).map(k=>`<div style="width:${cnt[k]/N*100}%;background:${COL[k]}"></div>`).join('');

  $('leg').innerHTML=`<span>Tỷ lệ sử dụng: <b style="color:var(--tx)">${Math.round(cnt.running/N*100)}%</b></span>`+Object.keys(ST).map(k=>`<span><i style="background:${COL[k]}"></i>${ST[k]} ${cnt[k]}</span>`).join('');

  $('alerts').textContent=low.length?'⚠ Nhiên liệu/pin thấp: '+low.join(', '):'';

  const card=v=>{const d=get(v.id),f=d.fuel??100;

    return `<button class="v ${d.status}" data-id="${v.id}">

    <div class="top"><span class="id">${full(v.id)}</span><span class="badge"><span class="dot"></span>${ST[d.status]}</span></div>

    <div class="op"><div class="av">${d.operator?esc(init(d.operator)):'–'}</div><div><b>${d.operator?esc(d.operator):'<span class="mu">Chưa phân công</span>'}</b><span>${d.task?esc(d.task):'—'}</span></div></div>

    <div class="kv"><span>Nhiên liệu/pin</span><b>${f}%</b></div><div class="meter ${f<=20?'low':f<=50?'mid':''}"><div style="width:${f}%"></div></div>

    <div class="kv"><span>Giờ máy</span><b>${live(d).toFixed(1)} giờ</b></div>

    <div class="kv"><span>${d.since&&d.status!=='off'?'Trạng thái được':'Trạng thái'}</span><b>${d.since&&d.status!=='off'?dur(Date.now()-d.since):'—'}</b></div></button>`};

  const list=t=>VEHICLES.filter(v=>v.type===t&&(filter==='all'||get(v.id).status===filter)).map(card).join('')||'<div class="mu">Không có xe phù hợp</div>';

  $('gRS').innerHTML=list('RS');$('gFL').innerHTML=list('FL');

}


document.addEventListener('click',e=>{

  const s=e.target.closest('.stat');if(s){filter=filter===s.dataset.f?'all':s.dataset.f;render();return}

  const c=e.target.closest('.v');if(c)openDlg(c.dataset.id)});

$('fFuel').oninput=()=>$('fuelV').textContent=$('fFuel').value;


function openDlg(id){cur=id;const d=get(id);

  $('dTitle').textContent=full(id);$('dInfo').textContent='Giờ máy tích lũy: '+live(d).toFixed(1)+' giờ';

  $('fStatus').value=d.status;$('fOp').value=d.operator;$('fTask').value=d.task;$('fFuel').value=d.fuel??100;$('fuelV').textContent=d.fuel??100;

  $('hist').innerHTML='<b style="font-size:13px">Lịch sử ca làm việc</b>'+((d.log||[]).slice().reverse().map(l=>`<div class="h"><time>${fmt(l.t)}</time><span><b style="color:${COL[l.s]}">${ST[l.s]}</b> · ${esc(l.o||'—')} · ${esc(l.k||'—')}</span></div>`).join('')||'<div class="mu" style="margin-top:6px">Chưa có lịch sử</div>');

  $('dlg').showModal()}


$('dCancel').onclick=()=>$('dlg').close();

$('dClear').onclick=()=>{$('fStatus').value='idle';$('fOp').value='';$('fTask').value='';$('dSave').click()};

$('dSave').onclick=async()=>{

  const old=get(cur),status=$('fStatus').value,operator=$('fOp').value.trim(),task=$('fTask').value.trim(),fuel=+$('fFuel').value,now=Date.now();

  const changed=status!==old.status||operator!==old.operator||task!==old.task;

  let hours=old.hours||0;if(changed&&old.status==='running'&&old.since)hours+=(now-old.since)/36e5;

  const log=(old.log||[]).slice(-29);if(changed)log.push({t:now,s:status,o:operator,k:task});

  const data={status,operator,task,fuel,hours,log,since:changed||!old.since?now:old.since};

  state[cur]=data;render();$('dlg').close();

  try{

    await store.save(cur,data);

    toast('Đã lưu và đồng bộ lên hệ thống');

  }catch(e){

    toast('Lỗi lưu dữ liệu: Hãy kiểm tra mục Rules trên Firebase');

  }

};


function toast(m){const t=$('toast');t.textContent=m;t.style.display='block';setTimeout(()=>t.style.display='none',3200)}

setInterval(()=>$('clock').textContent=new Date().toLocaleTimeString('vi-VN'),1000);

setInterval(()=>{if(!$('dlg').open)render()},30000);render();


/* Kết nối trực tiếp tới Firebase Realtime Database */

async function connectFirebase() {

  let activeUrl = FIREBASE_URLS[0];

  for (const url of FIREBASE_URLS) {

    try {

      const res = await fetch(url + "/vehicles.json", { method: "GET" });

      if (res.status !== 404) { activeUrl = url; break; }

    } catch (e) {}

  }


  const app = initializeApp({ databaseURL: activeUrl });

  const db = getDatabase(app);


  store = {

    save: (id, data) => set(ref(db, 'vehicles/' + id), data)

  };


  onValue(ref(db, 'vehicles'), (snapshot) => {

    const val = snapshot.val() || {};

    Object.keys(val).forEach(k => { state[k] = val[k]; });

    $('sync').textContent = 'Trực tiếp (Đã đồng bộ)';

    $('syncDot').style.background = 'var(--run)';

    if (!$('dlg').open) render();

  }, (error) => {

    console.error(error);

    $('sync').textContent = 'Chặn quyền truy cập (Kiểm tra Rules)';

    $('syncDot').style.background = 'var(--mt)';

  });

}


connectFirebase();
